import hashlib
import json
import os
from pathlib import Path
import resource
import signal
import subprocess
import sys
import time

ROOT = Path('/work/repo')
OUT = Path('/work/evidence80'); OUT.mkdir(exist_ok=True)
BASE = ['.venv/bin/python', '-B', '-m', 'pytest', '-q', '-p', 'no:cacheprovider']

def run(name, args):
    log = OUT / (name + '.log')
    with log.open('w') as stream:
        result = subprocess.run(args, cwd=ROOT, stdout=stream, stderr=subprocess.STDOUT, timeout=45)
    record = {'name':name, 'command':args, 'exit':result.returncode, 'output':log.read_text()}
    return record

def campaign():
    stage=os.environ['STAGE']
    row=run(stage,BASE+['tests/test_tier_budget.py'])
    hashes={name:hashlib.sha256((ROOT/name).read_bytes()).hexdigest() for name in ('tests/conftest.py','tests/test_tier_budget.py')}
    (OUT/(stage+'.json')).write_text(json.dumps({'parent':'fad2bc07aa90a96f7f273c5d11cbe01552b421af','sha256':hashes,**row},indent=2)+'\n')
    raise SystemExit(row['exit'])

if len(sys.argv)>1:
    resource.setrlimit(resource.RLIMIT_FSIZE,(8*1024**2,8*1024**2))
    campaign()
else:
    os.environ.update(TMPDIR='/tmp', PYTHONDONTWRITEBYTECODE='1', PATH='/work/bootstrap/bin:/usr/local/bin:/usr/bin:/bin')
    samples=[]
    proc=None
    started=time.monotonic()
    reason=None
    try:
        while True:
            usage=int(subprocess.check_output(['du','-s','-B1','/work','/tmp','/dev/shm'],text=True).splitlines()[0].split()[0])
            # Separate invocations preserve all mount totals and avoid parent suppression.
            usage += sum(int(subprocess.check_output(['du','-s','-B1',str(p)],text=True).split()[0]) for p in ('/tmp','/dev/shm'))
            logs=sum(p.stat().st_size for p in OUT.rglob('*') if p.is_file())
            free=os.statvfs('/work').f_bavail*os.statvfs('/work').f_frsize
            samples.append({'elapsed_s':round(time.monotonic()-started,2),'all_work_tmp_shm_bytes':usage,'evidence_bytes':logs,'vm_free_bytes':free})
            if usage>1024**3 or logs>8*1024**2 or free<50*1024**3 or time.monotonic()-started>240:
                reason='resource/deadline guard refused'; raise RuntimeError(reason)
            if proc is None:
                proc=subprocess.Popen([sys.executable,__file__,'campaign'],start_new_session=True)
            if proc.poll() is not None: break
            time.sleep(.5)
    finally:
        if proc is not None:
            # The leader may already be gone while its descendants are still alive.
            try:
                os.killpg(proc.pid,signal.SIGKILL)
            except ProcessLookupError:
                pass
            proc.wait()
        (OUT/(os.environ['STAGE']+'-resources.json')).write_text(json.dumps({'limits':{'all_work_tmp_shm_bytes':1024**3,'evidence_bytes':8*1024**2,'min_vm_free_bytes':50*1024**3,'deadline_s':240,'file_bytes':8*1024**2},'reason':reason,'samples':samples},indent=2)+'\n')
    raise SystemExit(proc.returncode)
